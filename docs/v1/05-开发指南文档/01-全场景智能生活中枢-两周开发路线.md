## 一、 项目概述

### 1.1 项目定位

```
小智（已有）+ 多传感器 + 场景引擎 + 多设备协同
= 全场景智能生活中枢
```

### 1.3 目标用户
- **老年人**：情感陪伴、跌倒急救、生活辅助
- **盲人**：空间导航、语音播报、环境感知
- **普通人**：智能联动、场景自动化、个性化服务

```c
#ifndef __BSP_MPU6050_H__
#define __BSP_MPU6050_H__

#include <stdint.h>
#include <stdbool.h>

typedef struct {
    float pitch;  // 俯仰角
    float roll;   // 横滚角
    float yaw;    // 偏航角
} mpu6050_attitude_t;

typedef struct {
    float accel_x, accel_y, accel_z;
    float gyro_x, gyro_y, gyro_z;
} mpu6050_raw_data_t;

typedef enum {
    MPU6050_STATE_NORMAL,
    MPU6050_STATE_FALLING,
    MPU6050_STATE_FALLEN
} mpu6050_state_t;

void BSP_MPU6050_Init(void);
bool BSP_MPU6050_Read_Raw(mpu6050_raw_data_t* data);
bool BSP_MPU6050_Get_Attitude(mpu6050_attitude_t* attitude);
mpu6050_state_t BSP_MPU6050_Detect_State(void);

#endif
```

**跌倒检测算法**：
```c
mpu6050_state_t BSP_MPU6050_Detect_State(void) {
    static mpu6050_attitude_t last_attitude;
    static uint32_t last_fall_time = 0;
    
    mpu6050_attitude_t current_attitude;
    if (!BSP_MPU6050_Get_Attitude(&current_attitude)) {
        return MPU6050_STATE_NORMAL;
    }
    
    // 检测剧烈角度变化（跌倒）
    float pitch_diff = fabs(current_attitude.pitch - last_attitude.pitch);
    float roll_diff = fabs(current_attitude.roll - last_attitude.roll);
    
    if (pitch_diff > 45.0f || roll_diff > 45.0f) {
        // 可能跌倒
        uint32_t now = xTaskGetTickCount();
        if (now - last_fall_time > 5000) {  // 5秒内不重复触发
            last_fall_time = now;
            return MPU6050_STATE_FALLEN;
        }
    }
    
    last_attitude = current_attitude;
    return MPU6050_STATE_NORMAL;
}
```

```c
// 学习用户起床时间
void App_Personalization_Learn_Habits(void) {
    // 记录用户每天起床时间
    static int wake_times[7] = {0};
    static int day_index = 0;
    
    // 检测到用户起床（姿态变化 + 语音）
    if (User_Wake_Up_Detected()) {
        wake_times[day_index] = Current_Hour();
        day_index = (day_index + 1) % 7;
        
        // 计算平均起床时间
        int avg_wake_time = 0;
        for (int i = 0; i < 7; i++) {
            avg_wake_time += wake_times[i];
        }
        avg_wake_time /= 7;
        
        // 保存到用户档案
        Save_Wake_Time_To_Profile(avg_wake_time);
    }
}

// 预测用户下一个行为
void App_Personalization_Predict_Next_Action(void) {
    user_profile_t* profile = Get_Current_User_Profile();
    
    // 如果接近睡眠时间，预测用户要睡觉
    int current_hour = Current_Hour();
    if (current_hour == profile->sleep_time_hour - 1) {
        // 提前准备睡眠模式
        AI_Response("快到睡觉时间了，需要我帮您准备吗？");
    }
}
```

```c
    uint8_t broadcast_mac[] = {0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF};

    // 等待响应（超时 5 秒）
    uint32_t start_time = xTaskGetTickCount();
    while (xTaskGetTickCount() - start_time < 5000) {
        // 处理接收到的设备信息...
    }
    
    ESP_LOGI(TAG, "发现 %d 个子设备", g_device_count);
    return true;
}

// 控制子设备
bool App_Device_Hub_Control(const char* device_name, const char* action) {
    for (int i = 0; i < g_device_count; i++) {
        if (strcmp(g_devices[i].device_name, device_name) == 0) {
            // 发送控制命令
            ESP_NOW_Send(g_devices[i].mac, action, strlen(action));
            return true;
        }
    }
    return false;
}
```
